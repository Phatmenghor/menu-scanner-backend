    FROM GENERATE_SERIES(1, 5000) AS gs(n);

    -- Add product images
    INSERT INTO product_images (

    SELECT
        gen_random_uuid(), 0, t, t, 'system', 'system', false, NULL, NULL,
        p.id, 'https://img.freepik.com/free-photo/food-image-' || ROW_NUMBER() OVER (PARTITION BY p.id ORDER BY p.id) || '.jpg',
        (ROW_NUMBER() OVER (PARTITION BY p.id ORDER BY p.id) = 1), ROW_NUMBER() OVER (PARTITION BY p.id ORDER BY p.id)
    FROM products p
    WHERE p.business_id = bid
    AND (ROW_NUMBER() OVER (PARTITION BY p.id ORDER BY p.id) <= 3);

    -- Add product sizes
    INSERT INTO product_sizes (
        id, version, created_at, updated_at, created_by, updated_by,

    )
    SELECT
        gen_random_uuid(), 0, t, t, 'system', 'system', false, NULL, NULL,
        p.id,
        CASE (ROW_NUMBER() OVER (PARTITION BY p.id ORDER BY p.id) % 3)

-- PHASE 5  –  ORDERS (10,000) WITH ITEMS & PAYMENTS

DO $$
DECLARE
    t TIMESTAMPTZ := NOW();

    cid UUID;
    pid UUID;
    sid UUID;
    customer_id UUID;
    order_id UUID;
    payment_status TEXT;
    order_counter INT := 0;
BEGIN
    SELECT id INTO bid FROM businesses WHERE email = 'phatmenghor20@gmail.com';

    FOR order_counter IN 1..10000 LOOP
        -- Random customer
        SELECT id INTO customer_id FROM users WHERE user_type = 'CUSTOMER' ORDER BY RANDOM() LIMIT 1;

        -- Generate order
        order_id := gen_random_uuid();
        payment_status := CASE WHEN order_counter % 3 = 0 THEN 'PENDING' ELSE 'COMPLETED' END;

        INSERT INTO orders (
            id, version, created_at, updated_at, created_by, updated_by,
            is_deleted, deleted_at, deleted_by,
            business_id, customer_id, total_amount, status, special_instructions,
            delivery_address, contact_phone, is_delivery, delivery_fee,
            discount_amount, tax_amount
        ) VALUES (
            order_id, 0, t - (RANDOM() * INTERVAL '60 days'), t,
            'system', 'system', false, NULL, NULL,
            bid, customer_id, (25 + (order_counter % 200))::NUMERIC,
            CASE WHEN order_counter % 5 = 0 THEN 'CANCELLED' WHEN order_counter % 4 = 0 THEN 'PENDING' ELSE 'COMPLETED' END,
            'Special instructions #' || order_counter,
            'Delivery Address ' || order_counter, '+855 10 123 4567', true, (2.5 + (order_counter % 5))::NUMERIC,
            (order_counter % 10)::NUMERIC, (order_counter % 5)::NUMERIC
        );

        -- Add 2-5 items per order
        FOR item_counter IN 1..(2 + (order_counter % 4)) LOOP
            SELECT id INTO pid FROM products WHERE business_id = bid ORDER BY RANDOM() LIMIT 1;
            SELECT id INTO sid FROM product_sizes WHERE product_id = pid LIMIT 1;

            INSERT INTO order_items (
                id, version, created_at, updated_at, created_by, updated_by,
                is_deleted, deleted_at, deleted_by,
                order_id, product_id, size_id, quantity, price, subtotal
            ) VALUES (
                gen_random_uuid(), 0, t, t, 'system', 'system', false, NULL, NULL,
                order_id, pid, sid, (1 + item_counter), (15 + (order_counter % 50))::NUMERIC,
                ((1 + item_counter) * (15 + (order_counter % 50)))::NUMERIC
            );
        END LOOP;

        -- Add order payment
        INSERT INTO order_payments (
            id, version, created_at, updated_at, created_by, updated_by,
            is_deleted, deleted_at, deleted_by,
            business_id, order_id, payment_reference, subtotal, discount_amount,
            delivery_fee, tax_amount, total_amount, payment_method, status, customer_payment_method
        ) VALUES (
            gen_random_uuid(), 0, t, t, 'system', 'system', false, NULL, NULL,
            bid, order_id, 'PAY-' || LPAD(order_counter::TEXT, 8, '0'),
            (25 + (order_counter % 200))::NUMERIC, (order_counter % 10)::NUMERIC,
            (2.5 + (order_counter % 5))::NUMERIC, (order_counter % 5)::NUMERIC,
            (25 + (order_counter % 200) - (order_counter % 10) + (2.5 + (order_counter % 5)) + (order_counter % 5))::NUMERIC,
            CASE (order_counter % 3) WHEN 0 THEN 'CARD' WHEN 1 THEN 'MOBILE' ELSE 'CASH' END,
            payment_status, 'Card'
        );

    END LOOP;

    RAISE NOTICE 'PHASE 5 Complete: 10,000 orders with items and payments created';

END $$;
