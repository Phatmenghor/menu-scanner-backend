-- PHASE 7  –  ORDER ITEMS
--             2-5 random ACTIVE products per order via LATERAL
-- ============================================================
INSERT INTO order_items (
    id, version, created_at, updated_at, created_by, updated_by,
    is_deleted, deleted_at, deleted_by,
    order_id, product_id, product_size_id,
    product_name, product_image_url, size_name,
    current_price, final_price, unit_price,
    has_promotion, promotion_type, promotion_value,
    quantity, total_price, special_instructions
)
SELECT
    gen_random_uuid(), 0, o.created_at, o.created_at,
    'system', 'system', false, NULL, NULL,
    o.id, p.id, NULL,
    p.name, p.main_image_url, NULL,
    p.price, p.price, p.price,
    p.has_active_promotion,
    CASE p.has_active_promotion WHEN true THEN p.promotion_type ELSE NULL END,
    CASE p.has_active_promotion WHEN true THEN p.promotion_value ELSE NULL END,
    qty.q,
    ROUND((p.price * qty.q)::NUMERIC, 2),
    NULL
FROM _seed_orders so
JOIN orders o ON o.id = so.id
CROSS JOIN LATERAL (
    SELECT id, name, main_image_url, price, has_active_promotion,
           promotion_type, promotion_value
    FROM products
    WHERE business_id = (SELECT val FROM _seed_ids WHERE key = 'business_id')
      AND status      = 'ACTIVE'
      AND is_deleted  = false
    ORDER BY RANDOM()
    LIMIT (2 + so.rn % 4)
) p
CROSS JOIN LATERAL (
    SELECT (1 + ABS(HASHTEXT(o.id::TEXT || p.id::TEXT)) % 4)::INTEGER AS q
) qty;
