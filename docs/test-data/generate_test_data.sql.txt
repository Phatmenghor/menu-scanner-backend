-- ============================================================
-- PHASE 5  –  ORDERS WITH ITEMS & PAYMENTS
-- ============================================================

DO $$
DECLARE
    t TIMESTAMPTZ := NOW();
    bid UUID;
    pid UUID;
    customer_id UUID;
    order_id UUID;
    total_amount NUMERIC;
BEGIN
    SELECT id INTO bid FROM businesses WHERE email = 'phatmenghor20@gmail.com';

    FOR order_counter IN 1..5000 LOOP
        SELECT id INTO customer_id FROM users WHERE user_type = 'CUSTOMER' ORDER BY RANDOM() LIMIT 1;
        SELECT id INTO pid FROM products WHERE business_id = bid ORDER BY RANDOM() LIMIT 1;

        order_id := gen_random_uuid();
        total_amount := (25 + (order_counter % 200))::NUMERIC;

        INSERT INTO orders (
            id, version, created_at, updated_at, created_by, updated_by, is_deleted, deleted_at, deleted_by,
            business_id, customer_id, total_amount, status, special_instructions,
            delivery_address, contact_phone, is_delivery, delivery_fee, discount_amount, tax_amount
        ) VALUES (
            order_id, 0, t - (RANDOM() * INTERVAL '60 days'), t, 'system', 'system', false, NULL, NULL,
            bid, customer_id, total_amount,
            CASE WHEN order_counter % 5 = 0 THEN 'CANCELLED' WHEN order_counter % 4 = 0 THEN 'PENDING' ELSE 'COMPLETED' END,
            'Instructions #' || order_counter, 'Address #' || order_counter, '+855 10 123 4567', true,
            (2.50 + (order_counter % 5))::NUMERIC, (order_counter % 10)::NUMERIC, (order_counter % 5)::NUMERIC
        );

        -- Add order items
        FOR item_n IN 1..(2 + (order_counter % 3)) LOOP
            INSERT INTO order_items (
                id, version, created_at, updated_at, created_by, updated_by, is_deleted, deleted_at, deleted_by,
                order_id, product_id, size_id, quantity, price, subtotal
            ) VALUES (
                gen_random_uuid(), 0, t, t, 'system', 'system', false, NULL, NULL,
                order_id, pid, NULL, (1 + item_n), (15 + (order_counter % 50))::NUMERIC,
                ((1 + item_n) * (15 + (order_counter % 50)))::NUMERIC
            );
        END LOOP;

        -- Add payment
        INSERT INTO order_payments (
            id, version, created_at, updated_at, created_by, updated_by, is_deleted, deleted_at, deleted_by,
            business_id, order_id, payment_reference, subtotal, discount_amount,
            delivery_fee, tax_amount, total_amount, payment_method, status, customer_payment_method
        ) VALUES (
            gen_random_uuid(), 0, t, t, 'system', 'system', false, NULL, NULL,
            bid, order_id, 'PAY-' || LPAD(order_counter::TEXT, 8, '0'),
            total_amount - (order_counter % 10)::NUMERIC, (order_counter % 10)::NUMERIC,
            (2.50 + (order_counter % 5))::NUMERIC, (order_counter % 5)::NUMERIC, total_amount,
            CASE (order_counter % 3) WHEN 0 THEN 'CARD' WHEN 1 THEN 'MOBILE' ELSE 'CASH' END,
            CASE WHEN order_counter % 3 = 0 THEN 'PENDING' ELSE 'COMPLETED' END, 'Card'
        );

    END LOOP;

    RAISE NOTICE 'PHASE 5 Complete: 5000 orders created';
END $$;
