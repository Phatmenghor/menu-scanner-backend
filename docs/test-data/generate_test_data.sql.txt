-- ============================================================
-- PHASE 0  –  CLEANUP  (idempotent — safe on empty or existing DB)
--             Deletes all previous test data in FK-safe order
-- ============================================================
DO $$
DECLARE
    v_bid  UUID;
    v_puid UUID;
    v_buid UUID;
    v_cuid UUID;
BEGIN
    -- Locate existing test records (NULL on first run)
    SELECT id INTO v_puid FROM users
        WHERE user_identifier = 'phatmenghor19@gmail.com' AND user_type = 'PLATFORM_USER';
    SELECT id INTO v_buid FROM users
        WHERE user_identifier = 'phatmenghor20@gmail.com' AND user_type = 'BUSINESS_USER';
    SELECT id INTO v_cuid FROM users
        WHERE user_identifier = 'phatmenghor21@gmail.com' AND user_type = 'CUSTOMER';
    SELECT id INTO v_bid  FROM businesses WHERE email = 'phatmenghor20@gmail.com';

    -- ── Auth / session data (delete BEFORE users) ─────────────
    DELETE FROM user_sessions WHERE user_id IN (
        SELECT id FROM users WHERE email LIKE 'staff%@phatrestaurant.com'
           OR email LIKE 'customer%@test.com'
           OR user_identifier IN ('phatmenghor19@gmail.com','phatmenghor20@gmail.com','phatmenghor21@gmail.com'));
    DELETE FROM refresh_tokens WHERE user_id IN (
        SELECT id FROM users WHERE email LIKE 'staff%@phatrestaurant.com'
           OR email LIKE 'customer%@test.com'
           OR user_identifier IN ('phatmenghor19@gmail.com','phatmenghor20@gmail.com','phatmenghor21@gmail.com'));
    DELETE FROM blacklisted_tokens
        WHERE user_identifier LIKE '%@phatrestaurant.com'
           OR user_identifier LIKE '%@test.com'
           OR user_identifier IN ('phatmenghor19@gmail.com','phatmenghor20@gmail.com','phatmenghor21@gmail.com');
    DELETE FROM audit_logs
        WHERE user_identifier LIKE '%@phatrestaurant.com'
           OR user_identifier LIKE '%@test.com'
           OR user_identifier IN ('phatmenghor19@gmail.com','phatmenghor20@gmail.com','phatmenghor21@gmail.com');

    -- ── Extra bulk users (staff + customers) ──────────────────
    DELETE FROM user_roles      WHERE user_id IN (SELECT id FROM users WHERE email LIKE 'staff%@phatrestaurant.com' OR email LIKE 'customer%@test.com');
    DELETE FROM customer_addresses WHERE user_id IN (SELECT id FROM users WHERE email LIKE 'customer%@test.com');
    DELETE FROM users           WHERE email LIKE 'staff%@phatrestaurant.com' OR email LIKE 'customer%@test.com';

    -- ── Business-scoped data ───────────────────────────────────
    IF v_bid IS NOT NULL THEN
        -- cart items for any cart belonging to this business
        DELETE FROM cart_items  WHERE cart_id IN (SELECT id FROM carts WHERE business_id = v_bid);
        DELETE FROM carts       WHERE business_id = v_bid;

        -- favourites, order detail, orders
        DELETE FROM product_favorites    WHERE product_id IN (SELECT id FROM products WHERE business_id = v_bid);
        DELETE FROM order_items          WHERE order_id   IN (SELECT id FROM orders   WHERE business_id = v_bid);
        DELETE FROM order_status_history WHERE order_id   IN (SELECT id FROM orders   WHERE business_id = v_bid);
        DELETE FROM orders               WHERE business_id = v_bid;

        -- product images / sizes / products
        DELETE FROM product_images WHERE product_id IN (SELECT id FROM products WHERE business_id = v_bid);
        DELETE FROM product_sizes  WHERE product_id IN (SELECT id FROM products WHERE business_id = v_bid);
        DELETE FROM products        WHERE business_id = v_bid;

        -- attendance check-ins, work days, order payments (before parent rows)
        DELETE FROM attendance_check_ins WHERE attendance_id IN (SELECT id FROM attendances  WHERE business_id = v_bid);
        DELETE FROM schedule_work_days   WHERE schedule_id   IN (SELECT id FROM work_schedules WHERE business_id = v_bid);
        DELETE FROM business_order_payments WHERE business_id = v_bid;

        -- HR data
        DELETE FROM attendances  WHERE business_id = v_bid;
        DELETE FROM leaves       WHERE business_id = v_bid;
        DELETE FROM work_schedules          WHERE business_id = v_bid;
        DELETE FROM work_schedule_type_enum WHERE business_id = v_bid;
        DELETE FROM leave_type_enum         WHERE business_id = v_bid;

        -- setup tables
        DELETE FROM banners                WHERE business_id = v_bid;
        DELETE FROM delivery_options       WHERE business_id = v_bid;
        DELETE FROM order_process_statuses WHERE business_id = v_bid;
        DELETE FROM business_settings      WHERE business_id = v_bid;
        DELETE FROM business_exchange_rates WHERE business_id = v_bid;
        DELETE FROM categories             WHERE business_id = v_bid;
        DELETE FROM brands                 WHERE business_id = v_bid;
        DELETE FROM notifications          WHERE business_id = v_bid;

        -- payments → subscriptions → plans (must be in this order)
        DELETE FROM payments       WHERE business_id = v_bid;
        DELETE FROM subscriptions  WHERE business_id = v_bid;
    END IF;

    -- global subscription plans (deleted after subscriptions)
    DELETE FROM subscription_plans
        WHERE name IN ('Annual Premium', 'Premium Monthly', 'Basic Monthly', 'Free Trial');

    -- global exchange rate seeded by this script
    DELETE FROM exchange_rates WHERE notes = 'Current USD to KHR rate';

    -- location reference data (deleted by code prefix)
    DELETE FROM location_village_cbc WHERE village_code  LIKE 'KH%';
    DELETE FROM location_commune_cbc  WHERE commune_code  LIKE 'KH%';
    DELETE FROM location_district_cbc WHERE district_code LIKE 'KH%';
    DELETE FROM location_province_cbc WHERE province_code LIKE 'KH%';

    -- seed placeholder images
    DELETE FROM images WHERE created_by = 'seed_script';

    -- ── Main customer cart / address / notifications ───────────
    IF v_cuid IS NOT NULL THEN
        DELETE FROM cart_items         WHERE cart_id IN (SELECT id FROM carts WHERE user_id = v_cuid);
        DELETE FROM carts              WHERE user_id = v_cuid;
        DELETE FROM customer_addresses WHERE user_id = v_cuid;
        DELETE FROM notifications      WHERE user_id = v_cuid;
    END IF;
    IF v_puid IS NOT NULL THEN DELETE FROM notifications WHERE user_id = v_puid; END IF;
    IF v_buid IS NOT NULL THEN DELETE FROM notifications WHERE user_id = v_buid; END IF;

    -- ── Business itself ────────────────────────────────────────
    IF v_bid IS NOT NULL THEN DELETE FROM businesses WHERE id = v_bid; END IF;

    -- ── Roles ─────────────────────────────────────────────────
    DELETE FROM user_roles WHERE user_id IN (v_puid, v_buid, v_cuid);
    DELETE FROM roles WHERE name = 'PLATFORM_ADMIN' AND user_type = 'PLATFORM_USER';
    DELETE FROM roles WHERE name = 'CUSTOMER'       AND user_type = 'CUSTOMER';
    IF v_bid IS NOT NULL THEN
        DELETE FROM roles WHERE name = 'BUSINESS_ADMIN' AND business_id = v_bid;
    END IF;

    -- ── Main 3 users ──────────────────────────────────────────
    DELETE FROM users WHERE id IN (v_puid, v_buid, v_cuid);

    RAISE NOTICE 'Cleanup complete — starting fresh seed.';
END $$;
